        { \bool_set_true:N \l_homework_image_row_numbered_bool }
      #6

        { \seq_count:N \l_homework_image_row_items_seq }

        { \tl_if_blank:nTF {#1} {\HomeworkImageRowDefaultScale} {#1} }
      \fp_set:Nn \l_homework_image_row_gap_ratio_fp
        { \tl_if_blank:nTF {#2} {\HomeworkImageRowDefaultGapRatio} {#2} }
      \dim_set:Nn \l_homework_image_row_max_height_dim
        { \tl_if_blank:nTF {#3} {\HomeworkImageRowDefaultMaxHeight} {#3} }

      \dim_set:Nn \l_homework_image_row_target_dim
        {
          \fp_to_dim:n
            {
              min(
                1,
                \l_homework_image_row_scale_fp *
                \l_homework_image_row_count_int /
                \HomeworkImageRowDefaultFullCount
              ) * \dim_to_fp:n {\linewidth}
            }
        }

      \dim_set:Nn \l_homework_image_row_image_area_dim
        {
          \fp_to_dim:n
            {
              (1 - \l_homework_image_row_gap_ratio_fp) *
              \dim_to_fp:n {\l_homework_image_row_target_dim}
            }
        }

      \int_compare:nNnTF {\l_homework_image_row_count_int} > {1}
        {
          \dim_set:Nn \l_homework_image_row_gap_dim
            {
              \fp_to_dim:n
                {
                  \l_homework_image_row_gap_ratio_fp *
                  \dim_to_fp:n {\l_homework_image_row_target_dim} /
                  (\l_homework_image_row_count_int - 1)
                }
            }
        }
        { \dim_zero:N \l_homework_image_row_gap_dim }

      \dim_zero:N \l_homework_image_row_ratio_sum_dim
      \cs_set_eq:NN
        \homework_image_row_item_data:nnn
        \homework_image_row_measure_item:nnn
      \seq_map_inline:Nn \l_homework_image_row_items_seq {##1}

      \dim_set:Nn \l_homework_image_row_height_dim
        {
          \fp_to_dim:n
            {
              \dim_to_fp:n {\l_homework_image_row_image_area_dim} /
              \dim_to_fp:n {\l_homework_image_row_ratio_sum_dim}
            }
        }
      \dim_compare:nNnT
        {\l_homework_image_row_height_dim} > {\l_homework_image_row_max_height_dim}
        {
          \dim_set_eq:NN
            \l_homework_image_row_height_dim
            \l_homework_image_row_max_height_dim
        }

      \bool_if:NTF \l_homework_image_row_numbered_bool
        {
          \refstepcounter{figure}
          \setcounter{homeworksubfigure}{0}
          \tl_if_blank:nF {#4} {\label{#4}}
          \cs_set_eq:NN
            \homework_image_row_item_data:nnn
            \homework_image_row_render_item:nnn
        }
        {
          \cs_set_eq:NN
            \homework_image_row_item_data:nnn
            \homework_image_row_render_bare_item:nnn
        }

      \makebox[\linewidth][c]{\hbox{
        \seq_map_indexed_inline:Nn \l_homework_image_row_items_seq
          {
            \bool_if:NT \l_homework_image_row_numbered_bool
              { \refstepcounter{homeworksubfigure} }
            ##2
            \int_compare:nNnT {##1} < {\l_homework_image_row_count_int}
              { \hspace{\dim_use:N \l_homework_image_row_gap_dim} }
          }
      }}

          \par\vspace{\abovecaptionskip}

            {\protect\numberline{\thefigure}#5}
        }

              \l__homework_measured_table_text_constraint_seq {##1}

              \l__homework_measured_table_width_seq
              \l__homework_measured_table_total_dim
            \dim_compare:nNnT

                    - \l__homework_measured_table_minimum_total_dim

                      \l__homework_measured_table_current_dim
                  }

                % minimum. Usually only the first eligible column is touched.
                \homework_measured_table_sum:NN
                  \l__homework_measured_table_width_seq
                  \l__homework_measured_table_total_dim
                \dim_set:Nn \l__homework_measured_table_excess_dim
                  {
                    \l__homework_measured_table_total_dim
                    - \l__homework_measured_table_available_dim
                  }
                \int_step_inline:nn
                  { \seq_count:N \l__homework_measured_table_width_seq }
                  {
                    \dim_compare:nNnT
                      { \l__homework_measured_table_excess_dim } > { 0pt }

                              \l__homework_measured_table_width_seq {##1}
                          }
                        \dim_set:Nn \l__homework_measured_table_slack_dim
                          {
                            \l__homework_measured_table_current_dim
                            - \seq_item:Nn
                                \l__homework_measured_table_minimum_seq {##1}
                          }
                        \dim_set:Nn
                          \l__homework_measured_table_reduction_dim
                          {
                            \dim_min:nn
                              { \l__homework_measured_table_excess_dim }
                              { \l__homework_measured_table_slack_dim }
                          }
                        \dim_sub:Nn \l__homework_measured_table_current_dim
                          { \l__homework_measured_table_reduction_dim }

                          { \l__homework_measured_table_reduction_dim }
                        \seq_set_item:NnV
                          \l__homework_measured_table_width_seq {##1}

      { \seq_count:N \l__homework_table_colspec_parts_seq - 1 }
    \int_compare:nNnT { \l__homework_table_column_count_int } > { 0 }
      {
        \dim_set:Nn \l__homework_table_available_width_dim { \linewidth }
        \dim_sub:Nn \l__homework_table_available_width_dim { 2\tabcolsep }
        \dim_sub:Nn \l__homework_table_available_width_dim { 2\doublerulesep }
        \int_step_inline:nn
          { \l__homework_table_column_count_int + 2 }
          {
            \dim_sub:Nn \l__homework_table_available_width_dim
              { \arrayrulewidth }
          }
      }
    \regex_replace_all:nnN {@\{\}} {} \l__homework_colspec_tl
    \bool_if:NTF \g__homework_measured_table_active_bool
      {
        % Each p{...} is only a Pandoc placeholder for a measured table.  Keep
        % its alignment prefix, discard the provisional width argument, and
        % insert the exact dimension calculated before longtable began.
        \seq_set_split:NnV \l__homework_table_colspec_parts_seq
          { p }
          \l__homework_colspec_tl
        \seq_pop_left:NN \l__homework_table_colspec_parts_seq
          \l__homework_rebuilt_colspec_tl

            \tl_trim_spaces:N \l__homework_colspec_part_tl
            \int_compare:nNnTF
              { \l_tmpa_int } < { \l__homework_table_column_count_int }
              {
                \exp_after:wN
                  \homework_colspec_discard_before_alignment:w
                  \l__homework_colspec_part_tl \q_stop
              }
              {
                \tl_clear:N \l__homework_colspec_tail_tl
              }
            \tl_put_right:Nx \l__homework_rebuilt_colspec_tl
              {
                M
                {
                  \seq_item:Nn
                    \g__homework_measured_table_width_seq
                    { \l_tmpa_int }
                }
                \exp_not:V \l__homework_colspec_tail_tl
              }
          }
        \tl_set_eq:NN \l__homework_colspec_tl
          \l__homework_rebuilt_colspec_tl
      }
      { \tl_replace_all:Nnn \l__homework_colspec_tl {p} {M} }
    \tl_if_in:NnTF \l__homework_colspec_tl {>}
      { \tl_replace_all:Nnn \l__homework_colspec_tl {>} {|>} }
      { \regex_replace_all:nnN {([lcr])} {|\1} \l__homework_colspec_tl }
    \tl_put_right:Nn \l__homework_colspec_tl {|}
    % Align longtable's left border with the hhline grid.
    \tl_put_left:Nn \l__homework_colspec_tl
      {@{\hspace{\dimexpr2\doublerulesep+\arrayrulewidth\relax}}}
    \tl_put_right:Nn \l__homework_colspec_tl {@{}}
  }

\cs_new_protected:Npn \homework_begin_longtable:nn #1#2
  {
    \dim_set:Nn \l__homework_table_original_linewidth_dim { \linewidth }
    \homework_prepare_colspec:n {#2}
    \bool_if:NTF \g__homework_measured_table_active_bool
      {
        % Pandoc writes multicolumn widths as fractions of \linewidth even
        % when the ordinary columns have been replaced with literal measured
        % dimensions. Match that reference length to the measured table's
        % complete outer width so spanning cells cannot retain page width.
        \homework_measured_table_sum:NN
          \g__homework_measured_table_width_seq
          \l__homework_table_measured_outer_width_dim
        \int_step_inline:nn
          { 2 * \l__homework_table_column_count_int }
          {
            \dim_add:Nn \l__homework_table_measured_outer_width_dim
              { \tabcolsep }
          }
        \dim_add:Nn \l__homework_table_measured_outer_width_dim
          { 2\doublerulesep }
        \int_step_inline:nn
          { \l__homework_table_column_count_int + 2 }
          {
            \dim_add:Nn \l__homework_table_measured_outer_width_dim
              { \arrayrulewidth }
          }
        \setlength{\linewidth}
          { \l__homework_table_measured_outer_width_dim }
      }
      {
        % Legacy Pandoc p-column expressions need the reduced local linewidth
        % before the template adds padding and grid rules.
        \int_compare:nNnT { \l__homework_table_column_count_int } > { 0 }
          {
            \setlength{\linewidth}
              { \l__homework_table_available_width_dim }
          }
      }
    \exp_args:NnV \homework_longtable_apply:nn {#1}
      \l__homework_colspec_tl
  }
